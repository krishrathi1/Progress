   513→  function escHtml(s) { return String(s).replace(/[&<>]/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;" }[c])); }
   514→  function escAttr(s) { return String(s).replace(/[&<>"]/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" }[c])); }
   515→  function hexA(hex, a) { if (hex.startsWith("var")) return hex; const h = hex.replace("#", ""); return `rgba(${parseInt(h.slice(0, 2), 16)},${parseInt(h.slice(2, 4), 16)},${parseInt(h.slice(4, 6), 16)},${a})`; }

   517→  /* ================= KEYBOARD ================= */
   518→  function onKey(e) {
   519→    const typing = /^(INPUT|TEXTAREA)$/.test(document.activeElement.tagName);
   520→    if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === "k") { e.preventDefault(); K.paletteOpen() ? K.closePalette() : openPalette(); return; }
   521→    if (typing) return;
   522→    if (e.key === "/") { e.preventDefault(); openPalette(); }
   523→    else if (e.key === "g") { location.hash = "#/dashboard"; }
   524→    else if (e.key === "a") { location.hash = "#/analytics"; }
   525→    else if (e.key === "f" && S.activeTimer) { openFocus(); }
   526→    else if (e.key === "?") { openShortcuts(); }
   527→  }
   528→  function openShortcuts() {

   532→      <div class="shortcuts">${keys.map(([k, d]) => `<div class="sc-row"><kbd>${k}</kbd><span>${d}</span></div>`).join("")}</div></div>`;
